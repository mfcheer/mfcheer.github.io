LightOJ 1038 - Race to 1 Again 【DP】

题意：题目很短，不叙述了。
解法：dp

#include <time.h>
#include <ctype.h>
#include <string.h>
#include <assert.h>

using namespace std;

int t, n;
double dp[100010];

int main()
{
    memset(dp, 0, sizeof(dp));
    dp[1] = 0;
    dp[2] = 2.0;
    double tmp;
    for (int i = 3; i <= 100010; i++)
    {
        tmp = 0;
        dp[i] = 0;
        for (int j = 1; j*j <= i; j++)
        {
            if (i % j == 0)
            {
                dp[i] += dp[j];
                tmp+=1;
                if (j != i / j && j != 1)
                {
                    dp[i] += dp[i / j];
                    tmp += 1;
                }
            }
        }
        dp[i] = (dp[i]  + tmp + 1) / tmp;
    }
    scanf("%d",&t);
    for (int ca = 1; ca <= t; ca++)
    {
        scanf("%d",&n);
        printf("Case %d: %.6lf\n",ca,dp[n]);
    }
    return 0;
}
