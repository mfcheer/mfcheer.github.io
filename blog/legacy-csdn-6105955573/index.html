LightOJ 1422 Halloween Costumes 【区间DP】

2015/06/18 00:05 · 272 字 · 约 1 分钟

版权声明：转载请注明出处。 https://blog.csdn.net/u014427196/article/details/46540437

题目链接： http://lightoj.com/volume_showproblem.php?problem=1422

解法：dp[i][j]=min(1+dp[i+1][j],dp[i+1][k-1]+dp[k][j])

代码：

#include <stdio.h>
#include <ctime>
#include <math.h>
#include <limits.h>
#include <complex>
#include <string>
#include <functional>
#include <iterator>
#include <algorithm>
#include <vector>
#include <stack>
#include <queue>
#include <set>
#include <map>
#include <list>
#include <bitset>
#include <sstream>
#include <iomanip>
#include <fstream>
#include <iostream>
#include <ctime>
#include <cmath>
#include <cstring>
#include <cstdio>
#include <time.h>
#include <ctype.h>
#include <string.h>
#include <assert.h>

using namespace std;

int c[110];
int t, n;
int dp[110][110];

int main()
{
    scanf("%d",&t);
    for (int ca = 1; ca <= t; ca++)
    {
        scanf("%d",&n);
        for (int i = 1; i <= n; i++)
            scanf("%d",&c[i]);

        memset(dp,0,sizeof(dp));
        for (int i = 1; i <= n; i++)
            for (int j = i; j <= n; j++)
                dp[i][j] = j - i + 1;

        for (int i = n-1; i >= 1; i--)
            for (int j = i + 1; j <= n; j++)
            {
                dp[i][j] = dp[i+1][j] + 1;//表示第i个的衣服在后面没有利用了
                for (int k = i+1; k <= j; k++)
                    if (c[i] == c[k])//用同一件衣服
                        dp[i][j] = min(dp[i][j],dp[i+1][k-1] + dp[k][j]);
            }

        printf("Case %d: %d\n",ca,dp[1][n]);
    }
    return 0;
}
