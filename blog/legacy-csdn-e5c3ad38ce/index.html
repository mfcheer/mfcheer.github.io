using namespace std;

int n;
int ar[100010];

int lowbit (int t)
{
    return  t&(-t);
}

void add (int i,int v)
{
    while (i<=n)
    {
        ar[i]+=v;
        i+=lowbit(i);
    }
}
int sum (int i)
{
    int ret =0 ;
    while (i>0)
    {
        ret += ar[i];
        i-=lowbit (i);
    }
    return ret;
}


int main()
{
    while (scanf ("%d",&n)!=EOF && n )
    {
        int a,b;
        memset (ar,0,sizeof (ar));
        int t=n;
        while (t--)
        {
            scanf ("%d%d",&a,&b);
            add(a,1);
            add(b+1,-1);
        }

        for (int i=1;i<n;i++)
            printf("%d ",sum(i));
            printf("%d\n",sum(n));
    }
    return 0;
}
