LightOJ 1044 - Palindrome Partitioning【dp】

2016/02/29 20:59 · 218 字 · 约 1 分钟

版权声明：转载请注明出处。 https://blog.csdn.net/u014427196/article/details/50768226

题目链接： http://www.lightoj.com/volume_showproblem.php?problem=1044

题意：给你一个字符串，问最少分为几个回文串？

思路：dp[i]表示从开头到位置 i 的最优解，若[j,i]是回文串，则dp[i] = min(dp[i],dp[j-1] +1)

代码：

#include <iostream>
#include <cstdio>
#include <algorithm>
#include <cmath>
#include <string.h>
#include <string>

using namespace std;

char s[1010];
int dp[1010];

bool is_ok(char a[], int L, int R)
{
    while (L <= R)
    {
        if (a[L] != a[R])
        {
            return false;
        }
        L++;
        R--;
    }

    return true;
}

int main()
{
    int T;
    scanf("%d", &T);
    for (int t = 1;t <= T;t++)
    {
        memset(dp, 0, sizeof(dp));

        scanf("%s", s + 1);
        int len = strlen(s+1);

        for (int i = 1;i <= len;i++)
        {
            dp[i] = 1010;
            for (int j = 1;j <= i;j++)
            {
                if (is_ok(s, j, i))
                {
                    if (j == 1)
                        dp[i] = 1;
                    else
                        dp[i] = min(dp[i], dp[j - 1] + 1);
                }
            }
        }

        printf("Case %d: %d\n", t, dp[len]);
    }
    return 0;
}
