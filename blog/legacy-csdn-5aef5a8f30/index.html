#include<iostream>

#include<ctype.h>
#include<algorithm>
#include<vector>
#include<string>
#include<queue>
#include<stack>
#include<set>
#include<map>

using namespace std;

int main()  
{  
    int n;  
    map<string,int> s;  
    while(cin>>n && n)
    {
        string str ;
        s.clear();
        for(int i=0;i<n;i++)
        {
            cin>> str;
            s[str]++;
        }

        map<string,int>::iterator r; 
        int max=-1;

        for(r=s.begin ();r != s.end();r++)
        {
            if(r->second > max)
                max=r ->second ;
        }
        for(r=s.begin ();r!=s.end();r++)
        {
            if(r->second ==max)
                cout<<r->first << endl; 
        }
    }

    return 0;  
}
