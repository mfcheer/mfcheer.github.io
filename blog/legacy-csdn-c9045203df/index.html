题目链接： http://codeforces.com/contest/558/problem/B

题意：给你一组数字。选择一个尽量短的区间，使其包含 数组中出现次数最多的元素的全部元素，求区间起始位置

代码：

#include <stdio.h>
#include <ctime>
#include <math.h>
#include <limits.h>
#include <complex>
#include <string>
#include <functional>
#include <iterator>
#include <algorithm>
#include <vector>
#include <stack>
#include <queue>
#include <set>
#include <map>
#include <list>
#include <bitset>
#include <sstream>
#include <iomanip>
#include <fstream>
#include <iostream>
#include <ctime>
#include <cmath>
#include <cstring>
#include <cstdio>
#include <time.h>
#include <ctype.h>
#include <string.h>
#include <assert.h>

using namespace std;

int n;
int l[1000010];//记录元素的起始位置
int r[1000010];

int p[1000010];//记录元素
int cnt[1000010];//记录元素出现的次数

int main()
{
    int n;
    while (scanf("%d", &n) != EOF)
    {
        memset(l, 0, sizeof(l));
        memset(r, 0, sizeof(r));
        memset(cnt, 0, sizeof(cnt));

        int res = 0;
        for (int i = 1;i <= n;i++)
        {
            scanf("%d",&p[i]);
            cnt[p[i]] ++;
            if (l[p[i]] == 0)
                l[p[i]] = i;
            r[p[i]] = i;
        }
        int MAX = 0;
        for (int i = 1;i <= n;i++)
            MAX = max(MAX, cnt[p[i]]);

        int t1, t2;
        int tmp = 1e6 + 10;

        for (int i = 1;i <= n;i++)
        {
            if (cnt[p[i]] == MAX)
                if (r[p[i]] - l[p[i]] < tmp)
                {
                    tmp = r[p[i]] - l[p[i]];
                    t1 = l[p[i]];
                    t2 = r[p[i]];
                }
        }
        printf("%d %d\n", t1, t2);
    }
    return 0;
}
