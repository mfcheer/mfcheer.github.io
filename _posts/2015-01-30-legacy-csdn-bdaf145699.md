把第一个相连的边设为0，其余的设为1.

    
    
    #include <iostream>
    #include <stdio.h>
    #include <string.h>
    #include <stdlib.h>
    #include <time.h>
    
    using namespace std;
    
    const int INF = 1000000;
    const int MAX = 110;
    
    int n,a,b;
    int m,f,t;
    int map[MAX][MAX];
    
    int main()
    {
        while (cin>>n>>a>>b)
        {
            for(int i=1;i<=n;i++)
                for(int j=1;j<=n;j++)
            {
                if (i == j)
                    map[i][j] = 0;
                else
                    map[i][j] = INF;
            }
            for(int i=1;i<=n;i++)
            {
              cin>> m;
              for(int j=1;j<=m;j++)
              {
                  cin>>t;
                  if (j==1)
                    map[i][t] = 0;
                else
                    map[i][t] = 1;
              }
            }
            for(int k=1;k<=n;k++)
                for(int i=1;i<=n;i++)
                    for(int j=1;j<=n;j++)
            {
                if (map[i][j] > map[i][k] + map[k][j])
                    map[i][j] = map[i][k] + map[k][j];
            }
    
            if (map[a][b] == INF)
                cout<<-1<<endl;
            else
                cout<<map[a][b]<<endl;
        }
        return 0;
    }
