    using namespace std;
    
    int n;
    int p[20][20];
    
    int vis[20];
    int k;
    int ans ;
    
    void dfs (int r,int val,int nu)
    {
        k=0;
        for(int i=1;i<=n;i++)
        {
            if (!vis[i] && (p[r][i] >= val))
            {
                vis[i] = 1;
                dfs(i,p[r][i],nu+1);
                vis[i] = 0;
                k = 1;
            }
        }
    
        if (!k)
            if (nu > ans)
                ans = nu;
    
        return ;
    }
    
    int main ()
    {
        while (scanf("%d",&n)!=EOF)
        {
            memset( p,0,sizeof(p) );
            memset(vis,0,sizeof(vis));
            ans = 0;
            for(int i=1;i<=n;i++)
                    for(int j=1;j<=n;j++)
                    {
                        scanf(" %d",&p[i][j]);
                    }
    
            vis[1] = 1;
            dfs(1,0,1);
    
            printf("%d\n",ans);
        }
        return 0;
    }
