    #include<math.h>
    #include<stdlib.h>
    #include<ctype.h>
    #include<algorithm>
    #include<vector>
    #include<string>
    #include<queue>
    #include<stack>
    #include<set>
    
    using namespace std;
    
    int n,m;
    char a[100][100];
    int num ,vis[100][100];
    
    void dfs(int i,int j)
    {
        if(!vis[i][j] && ( a[i][j]=='.' || a[i][j]=='@' ))
        {
            num++;
            vis[i][j]=1;
                     dfs(i-1,j);
            dfs(i,j-1);           dfs(i,j+1);
                     dfs(i+1,j);
        }
    }
    
    int main()
    {
        
        while(scanf("%d %d",&n,&m)!=EOF && n && m )
        {
            memset(a,'#',sizeof(a));
            for(int i=0;i<m;i++)
                    scanf("%s",a[i]);
    
            memset(vis,0,sizeof(vis));
            num=0;
    
            for(int i=0;i<m;i++)
                for(int j=0;j<n;j++)
                {
                    if(a[i][j]=='@')
                    {
                        dfs(i,j);
                    }
                }
                printf("%d\n",num);
        }
    }
