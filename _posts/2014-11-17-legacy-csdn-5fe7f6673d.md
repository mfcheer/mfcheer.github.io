    using namespace std;
    
    int n,m;
    int  p[1030];
    
    int main ()
    {
        cin>>m;
        while (m--)
        {
            cin>>n;
            for(int i=0;i<n;i++ )
                cin>>p[i];
    
            if ( n % 2 == 1 )
                p[n++]= 0;
            sort(p,p+n);
            int ans = 0;
            for (int i=0 ;i+1<n;i+=2)
            {
                ans ^= (p[i+1] - p[i] -1);
            }
            if (ans !=0)
                puts("Georgia will win");
            else
                puts("Bob will win");
        }
        return 0;
    }
