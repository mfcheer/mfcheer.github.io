题意：给你一个字符串，其中‘？’可替换为任何字符，问你是否可以替换？为字母使得字符串非回文。输出字典序最小的解。  
代码：

    
    
    #include <iostream>
    #include <stdio.h>
    #include <string>
    #include <algorithm>
    #include <string.h>
    #include <queue>
    #include <set>
    #include <map>
    #include <math.h>
    #include <stack>
    
    using namespace std;
    
    int n;
    char s[1010];
    int vis[1010];
    
    int solve()
    {
        memset(vis,0,sizeof(vis));
        if (s[n/2] =='?' && n%2 ==1)
            s[n/2] = 'a';
        for(int i=0;i<n;i++)
        {
            if (s[i] == '?')
                vis[i]=1;
        }
        for(int i=0;i<n;i++)
        {
            if (s[i]== '?')
                s[i]='a';
        }
        int ok=1;
        for(int i=0;i<=n/2;i++)
        {
            if (s[i] != s[n-i-1])
            {
                ok=0;
                break;
            }
        }
        if (ok)
        {
            for(int i=n-1;i>=1;i--)
            {
                if (vis[i]==1)
                {
                    s[i]+=1;
                    ok=0;
                    break;
                }
            }
        }
        if (ok)
            return 0;
        puts(s);
        return 1;
    }
    
    int main()
    {
        while (scanf("%d",&n)!=EOF)
        {
            scanf("%s",s);
            if (!solve())
                puts("QwQ");
        }
        return 0;
    }
