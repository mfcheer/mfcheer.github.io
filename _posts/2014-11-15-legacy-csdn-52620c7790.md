运用STL next_permutation 函数 ：

  

  

    
    
    #include<stdio.h>
    #include<iostream>
    #include<math.h>
    #include<stdlib.h>
    #include<ctype.h>
    #include<algorithm>
    #include<vector>
    #include<string.h>
    #include<queue>
    #include<stack>
    #include<set>
    #include<map>
    #include<sstream>
    #include<time.h>
    #include<utility>
    #include<malloc.h>
    #include<stdexcept>
    
    using namespace std;
    
    int  main ()
    {
        int n;
        scanf ("%d",&n);
        int a[10];
        for(int i=0 ;i<n;i++)
            cin>>a[i];
        sort (a,a+n);
        do
        {
            for(int i=0;i<n;i++)
                cout<<a[i]<<" ";
            cout<<endl;
        }while( next_permutation(a,a+n) );
        return 0;
    }
