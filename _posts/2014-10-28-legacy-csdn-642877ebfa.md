    using namespace std;
    int a[10000];
    int dp[10000];
    
    int maxx (int a,int b)
    {
        if (a>b)
            return a;
        else
            return b;
    }
    
    int main()
    {
        int n;
        while (scanf ("%d",&n)!=EOF && n )
        {
            for (int i=1;i<=n;i++)
            
                scanf ("%d",&a[i]);
    
            dp[1]=a[1];
            int ans=dp[1];
            int max ;
            for (int i=2;i<=n;i++)    
            {
                max=0;
                for (int j=1;j<i;j++)
                {
                    if (a[j]< a[i] && dp[j] > max )
                        max= dp[j];
                }
    
                dp[i]=max+a[i]; 
                ans= maxx (ans,dp[i]);
    
            }
            printf("%d\n",ans);
        }
        return 0;
    }
