        {
            for (int i=0;i<=n;i++)
                for (int j=0;j<=n;j++)
                {
                    d[i][j]=1000001;
                }
                
            int a,b,c;
    
            for (int i=0;i<m;i++)
            {
                scanf ("%d%d%d",&a,&b,&c);
                d[a][b]=d[b][a]=c;
            }
    
            for (int k=1;k<=n;k++)
                for (int i=1;i<=n;i++)
                    for (int j=1;j<=n;j++)
                    {
                        if (  d[i][j] > ( d[i][k]+d[k][j]) )
                            d[i][j] = ( d[i][k]+d[k][j] );
                    }
    
            printf ("%d\n",d[1][n]);
        }
        return 0;
    }
