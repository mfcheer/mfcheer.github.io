题意：一个整数a 和一个数组b，问你是否能在b中取出r个元素排列组成c数组满足a%c1%c1%…..%cr == 0。输出最小的r，不能满足条件输出-1。

思路：b按从大到小排序，暴搜。

代码：

    using namespace std;
    
    int n, a, b[300],ans;
    
    bool cmp11(int a, int b)
    {
        return a > b;
    }
    
    void dfs(int x,int num,int A)
    {
        if (x == 0)
        {
            ans = min(ans, A); return;
        }
        if (num == n) return;
    
        //printf("    %d\n",x);
        dfs(x % b[num + 1], num + 1, A + 1);
        dfs(x, num + 1, A);
    }
    
    int main()
    {
        //printf("3 % 7");
        int t;
        scanf("%d",&t);
        while (t--)
        {
            scanf("%d %d", &n, &a);
            for (int i = 1; i <= n; i++)
                scanf("%d", &b[i]);
            sort(b + 1, b + 1 + n, cmp11);
            ans = 500;
            dfs(a, 0, 0);
            if (ans == 500) ans = -1;
            printf("%d\n", ans);
        }
        return 0;
    }
