解法：dp[i][j]=min(1+dp[i+1][j],dp[i+1][k-1]+dp[k][j])

代码：

    using namespace std;
    
    int c[110];
    int t, n;
    int dp[110][110];
    
    int main()
    {
        scanf("%d",&t);
        for (int ca = 1; ca <= t; ca++)
        {
            scanf("%d",&n);
            for (int i = 1; i <= n; i++)
                scanf("%d",&c[i]);
    
            memset(dp,0,sizeof(dp));
            for (int i = 1; i <= n; i++)
                for (int j = i; j <= n; j++)
                    dp[i][j] = j - i + 1;
    
            for (int i = n-1; i >= 1; i--)
                for (int j = i + 1; j <= n; j++)
                {
                    dp[i][j] = dp[i+1][j] + 1;//表示第i个的衣服在后面没有利用了
                    for (int k = i+1; k <= j; k++)
                        if (c[i] == c[k])//用同一件衣服
                            dp[i][j] = min(dp[i][j],dp[i+1][k-1] + dp[k][j]);
                }
    
            printf("Case %d: %d\n",ca,dp[1][n]);
        }
        return 0;
    }
