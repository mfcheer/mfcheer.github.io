    using namespace std;
    
    int d[1110][1110];
    
    int main ()    
    {
        int n,m;
        while (scanf ("%d %d",&n,&m)!=EOF)
        {
            for (int i=0;i<n;i++)
                for (int j=0;j<n;j++)
                {
                    if (i==j)
                        d[i][j]=0;
                    else 
                        d[i][j]=1000010;
                }
    
            int a,b,c;
            for (int i=0;i<m;i++)
            {
                scanf ("%d%d%d",&a,&b,&c);
                if (d[a][b]>c)
                {
                    d[a][b]=d[b][a]=c;
                }
            }
    
            for (int k=0;k<n;k++)
                for (int i=0;i<n;i++)
                    for (int j=0;j<n;j++)
                    {
                        if (  d[i][j] > ( d[i][k]+d[k][j]) )
                            d[i][j] = ( d[i][k]+d[k][j] );
                    }
    
            int start ,end;
            scanf ("%d %d",&start ,&end);
    
            if (d[start][end]==1000010)
                printf("-1\n");
            else
                printf("%d\n",d[start][end]);
    
        }
        return 0;
    }
