题意：  
有n个瓶子，每个瓶子都有相应的水的数量，瓶子的水量可以乘2或者除2，求出最少多少步骤可以使瓶子中的所有水相等。

思路：  
将所有瓶子可以到达的毫升数及其步数记录起来，得到可以n个瓶子都能到达的毫升数的最小步数。

代码：

    
    
    #include <stdio.h>
    #include <ctime>
    #include <math.h>
    #include <limits.h>
    #include <complex>
    #include <string>
    #include <functional>
    #include <iterator>
    #include <algorithm>
    #include <vector>
    #include <stack>
    #include <queue>
    #include <set>
    #include <map>
    #include <list>
    #include <bitset>
    #include <sstream>
    #include <iomanip>
    #include <fstream>
    #include <iostream>
    #include <ctime>
    #include <cmath>
    #include <cstring>
    #include <cstdio>
    #include <time.h>
    #include <ctype.h>
    #include <string.h>
    #include <assert.h>
    
    using namespace std;
    
    int n;
    int cnt[100010];
    int vis[100010];
    int step[100010];
    
    int main()
    {
        while (scanf("%d", &n) != EOF)
        {
            memset(step,0,sizeof(step));
            memset(cnt, 0, sizeof(cnt)); 
            memset(vis, 0, sizeof(vis));
            queue <pair<int, int> > q;
    
            int s;
            for (int i = 1; i <= n; i++)
            {
                scanf("%d",&s);
                q.push(make_pair(s,0));
    
                while (!q.empty())
                {
                    int x = q.front().first;
                    int y = q.front().second;
                    q.pop();
                    if (x > 1e5 + 3) continue;
                    if (vis[x] == i) continue;
                    vis[x] = i;
                    cnt[x]++;
                    step[x] += y;
    
                    q.push(make_pair(x * 2, y + 1)); 
                    q.push(make_pair(x / 2, y + 1));
                }
            };
    
            int ans = 1e9;
            for(int i = 1; i <= 100005; i++)
            {
                if (cnt[i] == n)
                    ans = min(ans, step[i]);
            }
            printf("%d\n", ans);
        }
        return 0;
    }
