题意：求数组第k大的元素，直接排序会超时的，get到 nth_element 这个函数

nth_element 用法：

nth_element(start, start+n, end)

使第n大元素处于第n位置（从0开始,其位置是下标为n的元素），并且比这个元素小的元素都排在这个元素之前，比这个元素大的元素都排在这个元素之后，但不能保证他们是有序的,原理类似快速排序吧。

  

代码：

    
    
    #include <iostream>
    #include <stdio.h>
    #include <string>
    #include <string.h>
    #include <cmath>
    #include <queue>
    #include <vector>
    #include <map>
    #include <set>
    #include <algorithm>
    
    using namespace std;
    
    int a[5001000];
    int n, k;
    
    int main()
    {
        while (scanf("%d%d",&n,&k)!=EOF)
        {
            for(int i=1;i<=n;i++) scanf("%d",&a[i]);
            nth_element(a+1,a+1+n-k+1,a+1+n);
            printf("%d\n",a[n-k+1]);
        }
        return 0;
    }
