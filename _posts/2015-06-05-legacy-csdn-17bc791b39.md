问题：

把一个整数n分成k份，每份都不为0，有几种分法？

分析:

情况分为两类

一、

每份中不包含1的方法，为保证每份>=2，先拿出K个1到每一份，然后把剩下的n-k分为k份，分法有dp[n-k][k]

二、

至少有一份包含1，拿出1个当为一份。剩下的n-1个分为k-1份，分法有dp[n-1][k-1].

代码：

    
    
    #include <iostream>  
    #include <cstdio>  
    #include <cstring>  
    #include <algorithm>  
    #include <string>  
    #include <queue>  
    
    using namespace std;
    
    int n, k;
    int dp[222][10];
    
    int main()
    {
        while (cin >> n >> k)
        {
            dp[0][0] = 1;
            for (int i = 1; i <= n; i++)
            {
                for (int j = 1; j <= k; j++)
                {
                    if (i >= j) dp[i][j] = dp[i - j][j] + dp[i - 1][j - 1];
                }
            }
            cout << dp[n][k] << endl;
        }
        return 0;
    }
